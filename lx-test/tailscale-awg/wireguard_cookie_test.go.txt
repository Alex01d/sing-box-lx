// Synthetic test overlay; never changes the pinned submodule checkout.
package device

import (
 "strings"
 "testing"
 "time"
)

func TestAWGCookieHeadroomGateLX(t *testing.T) {
 for _,mode:=range []string{"awg2","awg3"} {
  t.Run(mode,func(t *testing.T){
   lines:="jc=3\njmin=10\njmax=50\ns1=32\ns2=32\ns3=32\ns4=16\nh1=1000-2000\nh2=3000-4000\nh3=5000-6000\nh4=7000-8000\ni1=<b 0xf6a1><r 16>\n"
   if mode=="awg3" {lines=strings.ReplaceAll(awg3DeviceLines(newHeaderKey(t)),"disable_cookies=1","disable_cookies=0")}
   pair,a,b:=newHeadroomDevicePair(t,lines)
   pair.devB.rate.underLoadUntil.Store(time.Now().Add(30*time.Second).UnixNano())
   packet:=buildIPv4Packet(testIPA,testIPB,64)
   send:=func(){pair.tunA.toDevice<-packet};send();awaitPacket(t,pair.tunB,packet,send)
   consumed:=false
   pair.devA.peers.RLock()
   for _,peer:=range pair.devA.peers.keyMap {peer.cookieGenerator.RLock();consumed=consumed||!peer.cookieGenerator.mac2.cookieSet.IsZero();peer.cookieGenerator.RUnlock()}
   pair.devA.peers.RUnlock()
   if !consumed {t.Fatal("application traffic passed without exercising a cookie reply")}
   for _,bind:=range []*headroomBind{a,b} { offsets:=bind.snapshot();if len(offsets)!=1||offsets[8]==0 {t.Fatalf("cookie/handshake/data headroom corrupted: %v",offsets)} }
  })
 }
}
